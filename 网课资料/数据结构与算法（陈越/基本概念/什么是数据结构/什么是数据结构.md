解决问题的效率跟==空间的利用效率==有关
```
#include<stdio.h>

//用循环实现
void printN(int n) {
	int i;
	for (i = 1; i <= n; i++) {
		printf("%d\n", i);
	}
}

//用递归实现
void printN_2(int n) {
	if (n) {
		printN(n - 1);
		printf("%d\n", n);
	}
}

int main(int argc, char* argv[])
{
	int n = 100000;
	printN(n);
	printN_2(n);
	return 0;
}
```
