#pragma warning(disable:4996)
#define _CRT_SECURE_NO_WARNINGS 1
#include <stdio.h>
#include <stdlib.h>

typedef struct treenode_ {
	int value;
	struct treenode_* left, * right;
	int flag;
}treeNode;

treeNode* createTree(int n);
treeNode* newNode(int number);
treeNode* insert(int number, treeNode* tree);
int check(treeNode* tree, int number);
int judge(treeNode* tree, int n);
void reset(treeNode* tree);
void freeTree(treeNode* tree);

int main() {
	int n, l, i;
	treeNode* tree;
	scanf("%d", &n);
	while (n) {
		scanf("%d", &l);
		tree = createTree(n);
		for (i = 0; i < l; i++) {
			if (judge(tree, n)) {
				printf("Yes\n");
			}
			else printf("No\n");
			reset(tree);
		}
		freeTree(tree);
		scanf("%d", &n);
	}

	return 0;
}

treeNode* createTree(int n) {
	treeNode* tree;
	int number,i;
	scanf("%d", &number);
	tree = newNode(number);
	for (i = 1; i < n; i++) {
		scanf("%d", &number);
		tree = insert(number,tree);
	}
	return tree;
}

treeNode* newNode(int number) {
	treeNode* p = (treeNode*)malloc(sizeof(treeNode));
	p->value = number;
	p->flag = 0;
	p->left = p->right = NULL;
	return p;
}

treeNode* insert(int number, treeNode* tree) {
	if (!tree) {
		tree = newNode(number);
	}
	else {
		if (number > tree->value)tree->right = insert(number,tree->right);
		else tree->left = insert(number,tree->left);
	}
	return tree;
}

int check(treeNode* tree, int number) {
	if (tree->flag) {
		if (number > tree->value)return check(tree->right, number);
		else if (number < tree->value)return check(tree->left, number);
		else return 0;
	}
	else {
		if (tree->value == number) {
			tree->flag = 1;
			return 1;
		}
		else return 0;
	}
}

int judge(treeNode* tree, int n) {
	int i, number, flag = 0;
	scanf("%d", &number);
	if (tree->value != number)flag = 1;
	else tree->flag = 1;
	for (i = 1; i < n; i++) {
		scanf("%d", &number);
		if ((!flag) && (!check(tree, number)))flag = 1;
	}
	if (flag)return 0;
	else return 1;
}

void reset(treeNode* tree) {
	tree->flag = 0;
	if (tree->left)reset(tree->left);
	if (tree->right)reset(tree->right);
}

void freeTree(treeNode* tree) {  //后序遍历
	if (tree->left)freeTree(tree->left);
	if (tree->right)freeTree(tree->right);
	free(tree);
}
