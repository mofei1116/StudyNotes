int main_1() {
	Node* head = NULL;//定义一个指针指向Node,初始化为NULL
	int number;
	do {
		scanf_s("%d", &number);
		if (number != -1) {
			//添加新的链表单元
			Node* p = (Node*)malloc(sizeof(Node));
			//把number加到新的链表单元中
			p->value = number;
			p->next = NULL;
			//通过遍历找到原链表最后一个
			Node* last = head;//从head开始遍历
			if (last) {//当head不指向NULL时（head不指向空链表即至少有一个链表单元）
				while (last->next) {//如果这个链表单元的next是NULL说明这是最后一个
					last = last->next;//聚焦到下一个链表单元
				}//while循环结束后last就是指向原链表最后一个单元的指针
				//连接
				last->next = p;
			}
			else head = p;
		}
	} while (number != -1);

	return 0;
}
