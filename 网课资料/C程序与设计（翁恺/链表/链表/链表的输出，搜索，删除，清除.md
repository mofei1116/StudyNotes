//自定义的数据结构来代表整个链表，具有可扩展性（可在List中扩充）
typedef struct _list {
	Node* head;
	//可定义一个Node* tail永远指向最后一个
}List;

void add(List* pList, int number);
void print(List* pList);
void sear(List* pList, int number);
void dele(List* pList, int number);
void free(List* pList);

int main() {
	List list;
	list.head=NULL;
	int number;
	do {
		scanf_s("%d", &number);
		if (number != -1) {
			add(&list, number);
		}
	} while (number != -1);
	print(&list);
	int n;
	scanf_s("%d", &n);
	dele(&list, n);
	print(&list);
	return 0;
}

//添加新的链表单元
void add(List* pList, int number) {
	Node* p = (Node*)malloc(sizeof(Node));
	//把number加到新的链表单元中
	p->value = number;
	p->next = NULL;
	//通过遍历找到原链表最后一个
	Node* last = pList->head;//从head开始遍历
	if (last) {//当head不指向NULL时（head不指向空链表即至少有一个链表单元）
		while (last->next) {//如果这个链表单元的next是NULL说明这是最后一个
			last = last->next;//聚焦到下一个链表单元
		}//while循环结束后last就是指向原链表最后一个单元的指针
		//连接
		last->next = p;
	}
	else pList->head = p;
}

//遍历并输出链表
void print(List* pList) {
	Node* p;
	for (p = pList->head; p; p = p->next) {
		printf("%d\t", p->value);
	}
	printf("\n");
}

//搜索链表
void sear(List* pList, int number) {
	int isFound = 0;
	Node* p;
	for (p = pList->head; p; p = p->next) {
		if (p->value == number) {
			printf("找到了\n");
			isFound = 1;
			break;
		}
	}
	if (!isFound) {
		printf("没找到\n");
	}
}

//删除链表单元
void dele(List* pList, int number) {
	Node* p;
	Node* q;
	for (q = NULL, p = pList->head; p; q = p, p = p->next) {
		if (p->value == number) {
			//防止边界情况，空指针不能在->的左边
			if (q) {
				q->next = p->next;
			}
			//q不存在（为NULL)，即为边界情况（number为链表中的第一个）
			else pList->head = p->next;
			free(p);
			break;
		}
	}
}

//清除链表
void free(List* pList) {
	Node* p;
	Node* q;
	for (p = pList->head; p; p=q) {
		q = p->next;
		free(p);
	}
}
