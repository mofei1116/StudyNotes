#include <stdio.h>

struct U0 {
	unsigned int leading : 3;
	unsigned int FLAG1 : 1;
	unsigned int FLAG2 : 1;
	int trailing : 27;
};

#include "prtBin.h"//prtBin为输出二进制的函数

int main()
{
	struct U0 uu;
	uu.leading = 2;
	uu.FLAG1 = 0;
	uu. FLAG2 = 1;
	uu.trailing = 0;
	printf("sizeof(uu)=%zu\n", sizeof(uu));
	prtBin(*(int*)&uu);/*&uu为uu的地址（指向一个结构体),
	再把这个指针强制转化为一个指向int型的指针,
	再调用这个地址的int型变量*/
	return 0;
}
